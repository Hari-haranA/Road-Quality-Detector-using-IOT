// Function to upload sensor readings to Firebase
void uploadSensorReadings(double DateTime ,float latitude, float longitude, float accZ, long Duration, long Distance) {
  // Prepare the sensor data to be uploaded as a JSON object
  String jsonPayload = "{";
  jsonPayload += "\"Detection DateTime\": " + String(DateTime, 12) + ",";
  jsonPayload += "\"Detection Latitude\": " + String(latitude, 6) + ",";
  jsonPayload += "\"Detection Longitude\": " + String(longitude, 6) + ",";
  jsonPayload += "\"Detection Acceleration\": " + String(accZ, 6) + ",";
  jsonPayload += "\"Detection Duration\": " + String(Duration, 6) + ",";
  jsonPayload += "\"Detection Distance\": " + String(Distance, 6);
  jsonPayload += "}";
  
  // Upload the data to Firebase
  if (Firebase.pushString(firebaseData, "/", jsonPayload)) {
    Serial.println("Sensor readings uploaded to Firebase successfully.");
  } else {
    Serial.println("Failed to upload sensor readings to Firebase.");
    Serial.println("Reason: " + firebaseData.errorReason());
  }
}

void loop() {
  //read gps data
  while(gps.available(serialGps)) {
   fix = gps.read();
    double Location_lat = fix.latitudeL();
    double Location_lon = fix.longitudeL();
       // read sensor data
      float ax = mpu6050.getAccX();
      float ay = mpu6050.getAccY();
      float az = mpu6050.getAccZ();
       // calculate Z-axis acceleration
       currentZAxis = az / 16384.0;
       // calculate rolling mean
       rollingMean = rollingMean * 0.98 + currentZAxis * 0.02;
       // calculate difference between current value and rolling mean
      difference = abs(currentZAxis - rollingMean);
       // check if anomaly is detected and acc threshold is 0.17
      if (difference > ACCZ_THRESHOLD) { 
         anomalyDetected = true;
        // check if bump is detected
            long duration1, distance1;
            digitalWrite(TRIGGER_PIN, LOW);
            delayMicroseconds(2);
            digitalWrite(TRIGGER_PIN, HIGH);
            delayMicroseconds(10);
            digitalWrite(TRIGGER_PIN, LOW);
            duration1 = pulseIn(ECHO_PIN, HIGH);
            distance1 = (duration1 / 2) / 29.1;
            AnomalyDistance = distance1;
            AnomalyDuration = duration1;
            if (distance1 < 30 && distance1 != 0) {
            bumpDetected = true;
             }    
        }
       // check if IR sensor detects sudden motion
       if (digitalRead(IR_PIN) == HIGH) {
         float irValue = digitalRead(IR_PIN);
         IRValue = irValue;
         irDetected = true;
         tone(BUZZER_PIN, 2000, 500); //play a tone with 1500 Frequency for 0.3 Second
         digitalWrite(RED_PIN, HIGH);
         delay(250);
         digitalWrite(RED_PIN, LOW);
         delay(250);
        }
     // if a bump was detected since the last reading, save the time and location to the EEPROM module
        if(detectedBump) {
            reading.t = (time_t)fix.dateTime;
            reading.Latitude = fix.latitudeL();
            reading.Longitude = fix.longitudeL();
            reading.accz = detectedAccZ;
            // check if bump is detected
            long duration, distance;
            digitalWrite(TRIGGER_PIN, LOW);
            delayMicroseconds(2);
            digitalWrite(TRIGGER_PIN, HIGH);
            delayMicroseconds(10);
            digitalWrite(TRIGGER_PIN, LOW);
            duration = pulseIn(ECHO_PIN, HIGH);
            distance = (duration / 2) / 29.1;
            if (distance < 30 && distance != 0) {
            bumpDetected = true;
            }
            uploadSensorReadings(reading.t,reading.Latitude / 1e7, reading.Longitude / 1e7, reading.accz, duration, distance);
            // reset detection variables for the rolling mean comparison
            detectedBump = false;
            detectedAccZ = 0;
           // flash the LED to signal a bump was stored  
         }
        
        // send data to Firebase if Wi-Fi is connected
       json.add("Calculated Raw Acceleration",currentZAxis);
       json.add("Threshold",ACCZ_THRESHOLD);
       json.add("Anomaly Duration", AnomalyDuration);
       json.add("Anomaly Distance", AnomalyDistance);
       json.add("Calculated Rolling Mean",rollingMean);
       json.add("Gps Driven Latitude",fix.latitudeL());
       json.add("Gps Driven Longitude",fix.longitudeL());
       json.add("Motion Detected",IRValue);
       if (Firebase.pushJSON(firebaseData, "/", json)) {
       Serial.println("Data uploaded to Firebase!");
       } else {
       Serial.println("Data upload failed...");
       Serial.println(firebaseData.errorReason());
       }
    delay(500);   
    }
  // add the current acceleration to the total to get an average acceleration every 20ms
    avgAZ += mpu6050.getAccZ();
    avgAZSamples++;
    
    // use the average acceleration every 20ms
    if(millis() - lastAvgAZ >= 20) {
        lastAvgAZ = millis();
        avgAZ /= avgAZSamples;
        
        // if the circle buffer is full, compare the current average acceleration with the current rolling mean of the data
        if(circleBufIndex >= ROLLING_WINDOW) {
            float relAmpAccZ = abs(avgAZ * ROLLING_WINDOW - circleBufSum) / 16384.0 / ROLLING_WINDOW;         // computationally cheap way to compute the rolling mean of the acceleration data
            
            // if the amplitude of the current acceleration is beyond a threshold away from the rolling mean, a bump was detected 
            if(relAmpAccZ > ACCZ_THRESHOLD && relAmpAccZ > detectedAccZ) {
                detectedAccZ = relAmpAccZ;
                detectedBump = true;
            } 
            
            // subtracts the last element of the circular buffer from the current sum of the data to keep computation speed fast
            circleBufSum -= circleBuf[circleBufIndex % ROLLING_WINDOW];
         }
        
        // append the average Z axis acceleration to the circle buffer for maintaining a rolling window of the baseline acceleration data
        circleBuf[circleBufIndex++ % ROLLING_WINDOW] = avgAZ;
        circleBufSum += avgAZ;

        // reset the average acceleration data for every 20ms
        avgAZ = 0;
        avgAZSamples = 0;
     }
  
}

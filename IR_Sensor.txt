#include <ESP8266WiFi.h>
#include <FirebaseArduino.h>

// Wi-Fi credentials
const char* ssid = "YourWiFiSSID";
const char* password = "YourWiFiPassword";

// Firebase configuration
#define FIREBASE_HOST "your-firebase-database-url"
#define FIREBASE_AUTH "your-firebase-authentication-secret"

// IR sensor pin
const int irPin = D3;

void setup() {
  Serial.begin(115200);

  // Connect to Wi-Fi
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(1000);
    Serial.println("Connecting to WiFi...");
  }
  Serial.println("Connected to WiFi!");

  // Initialize Firebase
  Firebase.begin(FIREBASE_HOST, FIREBASE_AUTH);
}

void loop() {
  // Read IR sensor
  int irValue = digitalRead(irPin);

  // Check if motion is detected
  if (irValue == HIGH) {
    // Motion detected
    Serial.println("Motion detected!");

    // Read other sensor values
    int sensorValue = analogRead(A0);

    // Create JSON object with sensor readings
    String json = "{\"motion\": true, \"sensorValue\": " + String(sensorValue) + "}";

    // Upload sensor readings to Firebase
    if (Firebase.push("/sensorData", json)) {
      Serial.println("Sensor data uploaded to Firebase!");
    } else {
      Serial.println("Failed to upload sensor data to Firebase.");
      Serial.println(Firebase.error());
    }
  }

  delay(1000); // Delay for 1 second
}
